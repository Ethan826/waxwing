effect Log { fn log(n: Int): Unit; };
fn pick(m: { h: Handler(Log with Console + ...e), ...r, ...s }): Handler(Log with Console + ...e) where r disjoint s = m.h;
fn relay(p: { h: Handler(Log with Console + ...e), ...r, ...s }): Handler(Log with Console + ...e) where r disjoint s = pick(p);
fn main(): Unit with Console = Console.print(1);
