effect Log { fn log(n: Int): Unit; };
effect Ask { fn ask(): Unit; };
fn mk(d: Unit -> Unit with ...s): Handler(nofail Ask with Log + ...s)
  with pure = handler Ask { ask() => Log.log(1) };
fn use(): Unit with nofail Log =
  with mk(fn(u: Unit) => ()) { defer Ask.ask(); () };
fn main(): Unit with Console =
  with handler Log { log(n) => Console.print(n) } { use() };
