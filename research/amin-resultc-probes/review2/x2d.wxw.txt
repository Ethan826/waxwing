type E = E;
effect Log { fn log(n: Int): Unit; };
type Th(...e) = Th(Unit -> Unit with ...e);
effect Ask { fn ask(): Unit; };
effect Get(a) { fn get(): a; };
fn mk(): Handler(Ask with Get(Th(...s)) + ...s) with pure =
  handler Ask { ask() => match Get.get() { Th(f) => f(()) } };
fn use(): Unit with Log =
  with handler Get(Th(nofail Log)) { get() => Th(fn(u: Unit) => { defer Log.log(7); () }) } {
    with mk() { Ask.ask() }
  };
fn main(): Unit with Console = handle {
  with handler Log { log(n) => Fail.fail(E) } { use() }
} { fail(error: E) => Console.print(99) };
