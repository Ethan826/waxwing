type E = E;
effect Log { fn log(n: Int): Unit; };
effect Box(a) { fn put(x: a): Unit; };
fn mk(): Handler(Box({ f: Unit -> Unit with ...s }) with ...s) with pure =
  handler Box({ f: Unit -> Unit with ...s }) { put(x) => x.f(()) };
fn use(): Unit with Log = with mk() { Box.put({ f: fn(u: Unit) => { defer Log.log(7); () } }) };
fn main(): Unit with Console = handle {
  with handler Log { log(n) => Fail.fail(E) } { use() }
} { fail(error: E) => Console.print(99) };
