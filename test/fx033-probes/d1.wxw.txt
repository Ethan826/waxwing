type E = E;
effect Log { fn log(n: Int): Unit; };
type Th(...e) = Th(Unit -> Unit with ...e);
effect Box(a) { fn put(x: a): Unit; };
fn mk(): Handler(Box(Th(...s)) with ...s) with pure =
  handler Box(Th(...s)) { put(x) => match x { Th(f) => f(()) } };
fn use(): Unit with Log = with mk() { Box.put(Th(fn(u: Unit) => { defer Log.log(7); () })) };
fn main(): Unit with Console = handle {
  with handler Log { log(n) => Fail.fail(E) } { use() }
} { fail(error: E) => Console.print(99) };
